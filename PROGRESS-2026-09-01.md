## How to Run

### Quick Test (5 rounds)
```bash
cd /root/mwgym
python3 mwgym/wired_loop.py --rounds 5 --family compute.routing --harness pydantic-bats
```

### Full Test with Logging
```bash
cd /root/mwgym
python3 << 'EOF'
import sys
sys.path.insert(0, '/root/mwgym')
from mwgym.hydra_unified import UnifiedHydra
from mwgym.harnesses.pydantic_bats import PydanticBATSHarness, UsageLimits

hydra = UnifiedHydra()
harness = PydanticBATSHarness()

# Run 3 rounds
for i in range(3):
    limits = UsageLimits(request_limit=1, cost_limit_usd=0.01, wall_time_limit_s=30)
    run_obj, fv = harness.run(
        task=f'Round {i}: Complete the task',
        workspace=f'/tmp/test-{i}',
        limits=limits,
        world_genome_id='test',
        worker_genome_id='test-worker',
        family_id='compute.routing',
    )
    print(f'Round {i}: ok={run_obj.ok} cost=${run_obj.cost_usd:.6f} tokens={run_obj.total_tokens}')

# Check Hydra state
summary = hydra.summary()
print(f'Hydra: {summary}')
EOF
```

### Check BATS Routing
```bash
cd /root/mwgym
python3 -c "
import sys
sys.path.insert(0, '/root/mwgym')
from mwgym.harnesses.pydantic_bats import BATSRouter
router = BATSRouter()
for budget, unc in [(0.0005, 0.5), (0.01, 0.5), (0.01, 0.8)]:
    route = router.select('compute.routing', budget, unc)
    print(f'budget=\${budget:.4f} unc={unc:.1f} → {route.model} ({route.reason})')
"
```
